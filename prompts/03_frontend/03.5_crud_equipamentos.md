# Prompt 03.5 — Gestão de Equipamentos no Painel Admin (CRUD)

> **Ordem de implementação:** Quinto sub-prompt do frontend. Implemente após 03.4.

---

## Objetivo

Implementar a tela completa de gestão do estoque de equipamentos (`AdminEquipamentos.tsx`) no painel administrativo, permitindo listar, cadastrar novo item, atualizar quantidade/dados e remover itens.

---

## Contexto

O backend expõe o CRUD completo de `/equipamentos` (Prompt 02.3) protegido por token de administrador. Esta tela fornece a interface visual para administração do laboratório.

---

## Escopo

Esta tarefa cobre **exclusivamente**:

- `frontend/src/pages/AdminEquipamentos.tsx` — tabela de estoque, botões de ação e modal/formulário para criação e edição.
- Registro da rota `/admin/equipamentos` no `App.tsx` (protegida com role `"admin"`).

---

## Restrições

> ⚠️ **Não altere nada além do que está descrito neste prompt.**
> Não altere telas de chat, autenticação de cliente ou endpoints do backend.

- Todas as chamadas para `/equipamentos` devem carregar o token de admin automaticamente via `api/client.ts`.
- O formulário deve validar que `quantidade >= 0`.
- Após criação, edição ou exclusão, a listagem deve ser atualizada imediatamente.

---

## Arquivo a Criar

### `frontend/src/pages/AdminEquipamentos.tsx`

```tsx
import React, { useEffect, useState } from 'react'
import { Link } from 'react-router-dom'
import api from '../api/client'
import { useAuth } from '../context/AuthContext'

interface Equipamento {
  id: number
  nome: string
  descricao: string | null
  categoria: string
  quantidade: number
}

export const AdminEquipamentos: React.FC = () => {
  const [equipamentos, setEquipamentos] = useState<Equipamento[]>([])
  const [carregando, setCarregando] = useState(true)
  const [modalAberto, setModalAberto] = useState(false)
  const [editandoId, setEditandoId] = useState<number | null>(null)

  const [nome, setNome] = useState('')
  const [descricao, setDescricao] = useState('')
  const [categoria, setCategoria] = useState('')
  const [quantidade, setQuantidade] = useState<number>(1)

  const { logout } = useAuth()

  const carregarEquipamentos = async () => {
    try {
      setCarregando(true)
      const res = await api.get('/equipamentos')
      setEquipamentos(res.data)
    } catch (err) {
      alert('Erro ao buscar equipamentos.')
    } finally {
      setCarregando(false)
    }
  }

  useEffect(() => {
    carregarEquipamentos()
  }, [])

  const abrirModalNovo = () => {
    setEditandoId(null)
    setNome('')
    setDescricao('')
    setCategoria('')
    setQuantidade(1)
    setModalAberto(true)
  }

  const abrirModalEditar = (eq: Equipamento) => {
    setEditandoId(eq.id)
    setNome(eq.nome)
    setDescricao(eq.descricao || '')
    setCategoria(eq.categoria)
    setQuantidade(eq.quantidade)
    setModalAberto(true)
  }

  const handleSalvar = async (e: React.FormEvent) => {
    e.preventDefault()
    if (quantidade < 0) {
      alert('A quantidade não pode ser negativa.')
      return
    }

    try {
      const payload = { nome, descricao: descricao || null, categoria, quantidade: Number(quantidade) }
      if (editandoId) {
        await api.put(`/equipamentos/${editandoId}`, payload)
      } else {
        await api.post('/equipamentos', payload)
      }
      setModalAberto(false)
      carregarEquipamentos()
    } catch (err: any) {
      alert(err.response?.data?.detail || 'Erro ao salvar equipamento.')
    }
  }

  const handleExcluir = async (id: number) => {
    if (!window.confirm('Tem certeza que deseja remover este equipamento do catálogo?')) return
    try {
      await api.delete(`/equipamentos/${id}`)
      setEquipamentos(equipamentos.filter((e) => e.id !== id))
    } catch (err: any) {
      alert(err.response?.data?.detail || 'Erro ao excluir equipamento.')
    }
  }

  return (
    <div style={{ padding: 24, maxWidth: 1100, margin: '0 auto' }}>
      <header style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 24 }}>
        <h1>Gestão de Estoque - Equipamentos</h1>
        <div>
          <Link to="/admin" style={{ marginRight: 16 }}>Ver Empréstimos</Link>
          <button onClick={logout}>Sair</button>
        </div>
      </header>

      <button onClick={abrirModalNovo} style={{ marginBottom: 16, padding: '8px 16px', background: '#16a34a', color: 'white', border: 'none', borderRadius: 4, cursor: 'pointer' }}>
        + Novo Equipamento
      </button>

      {carregando ? (
        <p>Carregando catálogo...</p>
      ) : (
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left' }}>
          <thead>
            <tr style={{ borderBottom: '2px solid #333' }}>
              <th style={{ padding: 8 }}>ID</th>
              <th style={{ padding: 8 }}>Nome</th>
              <th style={{ padding: 8 }}>Categoria</th>
              <th style={{ padding: 8 }}>Descrição</th>
              <th style={{ padding: 8 }}>Disponível</th>
              <th style={{ padding: 8 }}>Ações</th>
            </tr>
          </thead>
          <tbody>
            {equipamentos.map((eq) => (
              <tr key={eq.id} style={{ borderBottom: '1px solid #ddd' }}>
                <td style={{ padding: 8 }}>#{eq.id}</td>
                <td style={{ padding: 8 }}><strong>{eq.nome}</strong></td>
                <td style={{ padding: 8 }}>{eq.categoria}</td>
                <td style={{ padding: 8 }}>{eq.descricao || '-'}</td>
                <td style={{ padding: 8 }}>{eq.quantidade} un.</td>
                <td style={{ padding: 8 }}>
                  <button onClick={() => abrirModalEditar(eq)} style={{ marginRight: 8 }}>Editar</button>
                  <button onClick={() => handleExcluir(eq.id)} style={{ color: 'red' }}>Excluir</button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      )}

      {modalAberto && (
        <div style={{ position: 'fixed', inset: 0, background: 'rgba(0,0,0,0.5)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
          <div style={{ background: 'white', padding: 24, borderRadius: 8, width: 400 }}>
            <h3>{editandoId ? 'Editar Equipamento' : 'Novo Equipamento'}</h3>
            <form onSubmit={handleSalvar}>
              <div style={{ marginBottom: 12 }}>
                <label style={{ display: 'block' }}>Nome</label>
                <input style={{ width: '100%' }} required value={nome} onChange={(e) => setNome(e.target.value)} />
              </div>
              <div style={{ marginBottom: 12 }}>
                <label style={{ display: 'block' }}>Categoria</label>
                <input style={{ width: '100%' }} required value={categoria} onChange={(e) => setCategoria(e.target.value)} />
              </div>
              <div style={{ marginBottom: 12 }}>
                <label style={{ display: 'block' }}>Quantidade em Estoque</label>
                <input type="number" min="0" style={{ width: '100%' }} required value={quantidade} onChange={(e) => setQuantidade(Number(e.target.value))} />
              </div>
              <div style={{ marginBottom: 16 }}>
                <label style={{ display: 'block' }}>Descrição</label>
                <textarea style={{ width: '100%' }} value={descricao} onChange={(e) => setDescricao(e.target.value)} />
              </div>
              <div style={{ display: 'flex', justifyContent: 'flex-end', gap: 8 }}>
                <button type="button" onClick={() => setModalAberto(false)}>Cancelar</button>
                <button type="submit" style={{ background: '#2563eb', color: 'white' }}>Salvar</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  )
}
```
